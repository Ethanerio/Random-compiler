Random++ Language Reference

About Random++

Random++ is a programming language that uses Python syntax, compiled and executed entirely in your browser. It supports variables, functions, classes, list comprehensions, error handling, and much more.

Data Types

Random++ supports the following types:

# Numbers
x = 42          # int
y = 3.14        # float
z = 0xFF        # hex literal
b = 0b1010      # binary literal

# Strings
name = "Hello"
multi = """triple
quoted string"""
greeting = f"Hi, {name}!"

# Boolean & None
flag = True
empty = None

# Collections
my_list = [1, 2, 3]
my_dict = {"key": "value"}
my_set = {1, 2, 3}
my_tuple = (1, 2, 3)

Control Flow

# If/elif/else
if x > 0:
    print("positive")
elif x == 0:
    print("zero")
else:
    print("negative")

# While loop
while condition:
    do_something()

# For loop
for item in collection:
    process(item)

for i in range(10):
    print(i)

# Break and continue
for i in range(100):
    if i == 5:
        break
    if i % 2 == 0:
        continue
    print(i)

Functions

# Basic function
def greet(name):
    return f"Hello, {name}!"

# Default parameters
def power(base, exp=2):
    return base ** exp

# *args and **kwargs
def flexible(*args, **kwargs):
    print(args)
    print(kwargs)

# Lambda
square = lambda x: x ** 2

# Recursion
def factorial(n):
    if n <= 1:
        return 1
    return n * factorial(n - 1)

Classes

class Animal:
    def __init__(self, name, sound):
        self.name = name
        self.sound = sound

    def speak(self):
        return f"{self.name}: {self.sound}!"

class Dog(Animal):
    def __init__(self, name):
        self.name = name
        self.sound = "Woof"

dog = Dog("Rex")
print(dog.speak())

List Comprehensions

# Basic
squares = [x**2 for x in range(10)]

# With condition
evens = [x for x in range(20) if x % 2 == 0]

# Dict comprehension
scores = {name: len(name) for name in names}

# Ternary expression
result = "yes" if condition else "no"

Error Handling

try:
    risky_operation()
except Exception as e:
    print(f"Error: {e}")
finally:
    cleanup()

# Raise exceptions
raise "Something went wrong"

# Assert
assert x > 0, "x must be positive"
